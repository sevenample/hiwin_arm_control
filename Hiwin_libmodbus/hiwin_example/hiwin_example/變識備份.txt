import sys
import serial
import rclpy
from rclpy.node import Node
from PyQt5 import QtWidgets, QtCore
from hiwin_msgs.msg import CatchArray

class ShapeClassifier(Node):
    def __init__(self, update_gui_callback):
        super().__init__('shape_classifier')
        self.ser = serial.Serial('/dev/ttyUSB0', 9600, timeout=1)
        self.publisher_ = self.create_publisher(CatchArray, 'detected_shapes', 10)
        self.timer = self.create_timer(0.1, self.read_serial_data)
        self.update_gui_callback = update_gui_callback

        self.shape_code_map = {
            '大立方體': 'A', '中立方體': 'B', '小立方體': 'C',
            '圓柱': 'D', '三角柱（躺）': 'E', '三角柱（立）': 'E',
            '六角柱（立）': 'F', '六角柱（躺）': 'F',
            '大長方體': 'G', '小長方體': 'G', '小長方體（立）': 'G',
            '小長方體（立19）': 'G', '無法分類': 'NONE'
        }

        self.code_to_chinese = {
            'A': '大立方體', 'B': '中立方體', 'C': '小立方體',
            'D': '圓柱', 'E': '三角柱', 'F': '六角柱',
            'G': '長方體（異常）', 'NONE': '無法分類'
        }

    def read_serial_data(self):
        if self.ser.in_waiting > 0:
            line = self.ser.readline().decode('utf-8').strip()
            try:
                adc, b1, b2, b3, b4 = map(int, line.split(','))
                code = self.classify_shape(adc, b1, b2, b3, b4)
                label = self.code_to_chinese.get(code, '未知')

                msg = CatchArray()
                msg.items = [code]
                self.publisher_.publish(msg)

                self.update_gui_callback(f"檢測到：{label} \n（代碼：{code}）")
                self.get_logger().info(f"ADC={adc}, Buttons=({b1},{b2},{b3},{b4}) -> Code: {code} ({label})")
            except Exception as e:
                self.get_logger().warn(f"解析錯誤: {e} -> '{line}'")

    def classify_shape(self, adc, b1, b2, b3, b4):
        if adc >= 1000:
            return 'A' if b1 == 0 else 'G'
        elif adc >= 870:
            return 'F'
        elif 670 <= adc <= 820:
            return 'B' if b2 == 0 else 'E'
        elif 480 <= adc <= 660:
            if b4 == 0:
                return 'G'
            elif b1 == 0 and b3 != 0:
                return 'G'
            elif b2 == 0 and b3 == 0 :
                return 'F'
            else:
                return 'D'
        elif 380 <= adc <= 460:
            return 'E'
        elif 200 <= adc <= 370:
            return 'G'
        elif 35 <= adc <= 150:
            return 'C'
        elif adc < 30:
            return 'G'
        else:
            return 'NONE'

class ShapeGUI(QtWidgets.QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("物件辨識即時顯示")
        self.resize(400, 200)
        self.layout = QtWidgets.QVBoxLayout()

        self.label = QtWidgets.QLabel("等待資料中...", self)
        self.label.setAlignment(QtCore.Qt.AlignCenter)
        self.label.setStyleSheet("font-size: 128px; color: black;")
        self.layout.addWidget(self.label)

        self.setLayout(self.layout)

    def update_display(self, text):
        self.label.setText(text)

def main(args=None):
    rclpy.init(args=args)
    app = QtWidgets.QApplication(sys.argv)

    gui = ShapeGUI()

    # Create node with GUI callback
    node = ShapeClassifier(update_gui_callback=gui.update_display)

    # ROS2 + Qt 整合執行 loop
    def ros_spin_once():
        rclpy.spin_once(node, timeout_sec=0)

    timer = QtCore.QTimer()
    timer.timeout.connect(ros_spin_once)
    timer.start(10)

    gui.show()
    app.exec_()

    node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
